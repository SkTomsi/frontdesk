SEC-01Build Phases
Three Phases
Same phased approach as the Python build. Phase 1 gets you a working Agentic RAG on the Node stack. Phase 2 makes it observable and trustworthy. Phase 3 upgrades the quality ceiling. Each phase ships something testable end-to-end.

  • Parent chunk fetch from PostgreSQL via Prisma

  • End of week: Full Q&A pipeline working end-to-end. Ask a question, get a cited answer.

  • Add Langfuse to docker-compose.yml, configure self-hosted
  • Wrap every LangGraph.js node with a Langfuse span helper

  • Look at eval scores — pick the lowest dimension
  • If context precision low → semantic chunking
  • If faithfulness low on short queries → add HyDE

SEC-07API Contract
API Endpoints
Identical contract to the Python build. All endpoints require a Logto-issued access token via Authorization: Bearer <token> (except /health). The token is verified using jose against the JWKS at {LOGTO_ENDPOINT}/oidc/.well-known/jwks.json (cached via createRemoteJWKSet) with audience LOGTO_RESOURCE. The tenantId is always extracted server-side from the organization_id claim — never from the request body.
MethodEndpointDescriptionReturns
POST /ingest Upload a PDF (multipart via @fastify/multipart). Validates, deduplicates, dispatches a BullMQ job. Returns immediately — processing is async. {"jobId": "...", "status": "queued"}
GET /ingest/status/{jobId} Poll BullMQ job status. Returns: waiting | active | completed | failed with error detail. {"status": "completed", "chunkCount": 84}
POST /chat/stream Submit a query. Returns an SSE stream over Fastify's raw response: status events → token events → done event with full citations JSON. SSE stream
POST /feedback Submit thumbs up/down for a query. Body: {"traceId": "...", "rating": 1}. Wired to the Langfuse trace. Phase 2. {"status": "recorded"}
GET /documents List all active documents for the current tenant. Returns metadata only, not chunk content. [{"id": "...", "filename": "...", "version": 2}]
DELETE /documents/{docId} Soft-delete document — sets isActive=false on all Qdrant points and PostgreSQL parent chunks. Does not delete the raw file. {"status": "deactivated"}
GET /health Health check. Returns status of all downstream services: Postgres, Qdrant, Redis. No auth required. {"status": "ok", "services": {...}}
SSE Stream Format — /chat/stream sends three event types, written directly to reply.raw with Content-Type: text/event-stream:
event: status · data: {"node": "retrieve", "message": "Searching 84 chunks..."}
event: token · data: {"text": "According to"}
event: done · data: {"answer": "...", "citations": [...], "traceId": "...", "faithfulness": 0.94}

src/ingestion/queue.ts — BullMQ worker
TypeScript
import { Worker, Job } from "bullmq";
import { redisConnection } from "../core/queueClient";
import { parsePdfWithLlamaParse } from "./parser";
import { hierarchicalChunk } from "./chunker";
import { embedChunks } from "./embedder";
import { upsertToQdrant } from "./upserter";
import { writeParentChunks, validateChunkCount } from "../db/queries";

interface IngestJobData {
  filePath: string;
  tenantId: string;
  documentId: string;
}

export const ingestWorker = new Worker<IngestJobData>(
  "ingest",
  async (job: Job<IngestJobData>) => {
    const { filePath, tenantId, documentId } = job.data;

    // 1. Parse PDF with LlamaParse (tables → markdown, OCR for scanned)
    const pages = await parsePdfWithLlamaParse(filePath);

    // 2. Hierarchical chunking: parent 1024t → child 256t
    const { parentNodes, childNodes } = hierarchicalChunk(pages);

    // 3. Embed child chunks (dense + sparse)
    const embeddedChildren = await embedChunks(childNodes);

    // 4. Write parent chunks to PostgreSQL via Prisma
    await writeParentChunks(parentNodes, tenantId, documentId);

    // 5. Upsert children to Qdrant with named vectors
    const collection = `tenant_${tenantId}`;
    await upsertToQdrant(embeddedChildren, collection, documentId);

    // 6. Validate chunk count
    await validateChunkCount(collection, documentId, childNodes.length);
  },
  { connection: redisConnection, concurrency: 2, attempts: 3 }
);
src/ingestion/upserter.ts — Qdrant upsert with named vectors
TypeScript
import { qdrant } from "../core/qdrantClient";

interface EmbeddedChunk {
  nodeId: string; tenantId: string; docId: string; parentId: string;
  docType: string; pageNum: number; index: number; text: string;
  dense: number[]; sparse: { indices: number[]; values: number[] };
}

function buildPoint(chunk: EmbeddedChunk) {
  return {
    id: chunk.nodeId,
    vector: {
      dense: chunk.dense,              // 1536-dim, openai/text-embedding-3-small via OpenRouter
      bm25: { indices: chunk.sparse.indices, values: chunk.sparse.values } // @qdrant/fastembed sparse
    },
    payload: {
      tenant_id: chunk.tenantId,
      document_id: chunk.docId,
      parent_id: chunk.parentId,        // key for parent fetch
      doc_type: chunk.docType,
      page_num: chunk.pageNum,
      chunk_index: chunk.index,
      is_active: true,                // versioning flag
      text_preview: chunk.text.slice(0, 200)
    }
  };
}

export async function upsertToQdrant(chunks: EmbeddedChunk[], collection: string, documentId: string) {
  await qdrant.upsert(collection, { points: chunks.map(buildPoint) });
}

import { Annotation } from "@langchain/langgraph";
import { BaseMessage } from "@langchain/core/messages";

export const AgentState = Annotation.Root({
  // ── Core query ──────────────────────────────
  query: Annotation<string>,
  tenantId: Annotation<string>,
  collection: Annotation<string>,                     // Qdrant collection name
  queryType: Annotation<"simple_factual" | "multi_part" | "procedural">,
  subQuestions: Annotation<string[]>,                    // decomposed sub-queries
  iteration: Annotation<number>({ reducer: (_, next) => next, default: () => 0 }),

  // ── Retrieval ───────────────────────────────
  retrievedChunks: Annotation<Record<string, unknown>[]>,     // post-rerank top-8
  parentChunks: Annotation<Record<string, unknown>[]>,       // 1024-token context
  contextScore: Annotation<number>,                        // Gemini assessment 0-10
  contextReason: Annotation<string>,                       // used for reformulation

  // ── Generation ──────────────────────────────
  finalAnswer: Annotation<string>,
  citations: Annotation<Record<string, unknown>[]>,
  faithfulnessScore: Annotation<number>,                 // Phase 2

  // ── LangGraph ───────────────────────────────
  messages: Annotation<BaseMessage[]>({
    reducer: (left, right) => left.concat(right),
    default: () => []
  })
});

export type AgentStateType = typeof AgentState.State;
src/agent/graph.ts — LangGraph.js StateGraph wiring
TypeScript
import { StateGraph, END } from "@langchain/langgraph";
import { AgentState, AgentStateType } from "./state";
import {
  classifyQuery, decomposeQuestion, retrieve,
  rerank, parentFetch, assessContext,
  reformulateQuery, generateAnswer, faithfulnessCheck
} from "./nodes";

export function buildGraph() {
  const g = new StateGraph(AgentState)
    // ── Add nodes ──────────────────────────────
    .addNode("classify", classifyQuery)
    .addNode("decompose", decomposeQuestion)
    .addNode("retrieve", retrieve)
    .addNode("rerank", rerank)
    .addNode("parentFetch", parentFetch)
    .addNode("assess", assessContext)
    .addNode("reformulate", reformulateQuery)
    .addNode("generate", generateAnswer)
    .addNode("faithfulness", faithfulnessCheck)  // Phase 2

    // ── Entry ──────────────────────────────────
    .addEdge("__start__", "classify")

    // ── Classify routing ───────────────────────
    .addConditionalEdges("classify", (s: AgentStateType) => s.queryType, {
      multi_part: "decompose",
      simple_factual: "retrieve",
      procedural: "retrieve"
    })
    .addEdge("decompose", "retrieve")
    .addEdge("retrieve", "rerank")
    .addEdge("rerank", "parentFetch")
    .addEdge("parentFetch", "assess")

    // ── Assess routing ─────────────────────────
    .addConditionalEdges("assess", (state: AgentStateType) => {
      if (state.contextScore >= 7) return "generate";
      if (state.iteration < 3) return "reformulate";
      return "generate";             // best-effort after max iterations
    })
    .addEdge("reformulate", "retrieve")    // loop back
    .addEdge("generate", "faithfulness")
    .addEdge("faithfulness", END);

  return g.compile();
}

Hybrid Retrieval
src/retrieval/hybridSearch.ts — Qdrant prefetch + RRF
TypeScript
import { qdrant } from "../core/qdrantClient";
import { embedQueryDense } from "../ingestion/embedder";
import { embedQuerySparse } from "../ingestion/embedder";

export async function hybridSearch(
  queryText: string,
  tenantId: string,
  collection: string,
  limit = 30
) {
  // Embed query — dense
  const denseVec = await embedQueryDense(queryText);
  // Embed query — sparse (@qdrant/fastembed BM25)
  const sparseVec = await embedQuerySparse(queryText);

  // Qdrant tenant + version filter
  const tenantFilter = {
    must: [
      { key: "tenant_id", match: { value: tenantId } },
      { key: "is_active", match: { value: true } }
    ]
  };

  const results = await qdrant.query(collection, {
    prefetch: [
      { query: denseVec, using: "dense", filter: tenantFilter, limit },
      { query: sparseVec, using: "bm25", filter: tenantFilter, limit }
    ],
    query: { fusion: "rrf" },              // Reciprocal Rank Fusion
    limit,
    with_payload: true
  });

  return results.points;
}
src/retrieval/cache.ts — Redis semantic cache (ioredis)

import Redis from "ioredis";
import { cosineSimilarity } from "../core/vectorMath";
import { embedQueryDense } from "../ingestion/embedder";
import { config } from "../core/config";

const cacheRedis = new Redis(config.SEMANTIC_CACHE_REDIS_URL);

export async function getCachedAnswer(query: string, tenantId: string) {
  const queryVec = await embedQueryDense(query);
  const keys = await cacheRedis.keys(`cache:${tenantId}:*`);

  for (const key of keys) {
    const raw = await cacheRedis.get(key);
    if (!raw) continue;
    const entry = JSON.parse(raw);
    if (cosineSimilarity(queryVec, entry.vector) >= config.CACHE_SIMILARITY_THRESHOLD) {
      return entry.answer;
    }
  }
  return null;
}

export async function setCachedAnswer(query: string, tenantId: string, vector: number[], answer: unknown) {
  const key = `cache:${tenantId}:${crypto.randomUUID()}`;
  await cacheRedis.set(key, JSON.stringify({ vector, answer }), "EX", config.CACHE_TTL_SECONDS);
}

import { AutoTokenizer, AutoModelForSequenceClassification } from "@huggingface/transformers";

let tokenizer: Awaited<ReturnType<typeof AutoTokenizer.from_pretrained>>;
let model: Awaited<ReturnType<typeof AutoModelForSequenceClassification.from_pretrained>>;

async function loadReranker() {
  if (model) return;
  tokenizer = await AutoTokenizer.from_pretrained("BAAI/bge-reranker-v2-m3");
  model = await AutoModelForSequenceClassification.from_pretrained(
    "BAAI/bge-reranker-v2-m3", { dtype: "fp32" }
  );
}

export async function rerankChunks(query: string, chunks: { text: string }[], topK = 8) {
  await loadReranker();

  const scores: number[] = [];
  for (const chunk of chunks) {
    const inputs = await tokenizer(query, { text_pair: chunk.text, padding: true, truncation: true });
    const { logits } = await model(inputs);
    scores.push(logits.data[0]as number);
  }

  return chunks
    .map((chunk, i) => ({ chunk, score: scores[i] }))
    .sort((a, b) => b.score - a.score)
    .slice(0, topK)
    .map(r => r.chunk);
}

Faithfulness Check Node
src/agent/nodes/faithfulness.ts — post-generation verification

import { AgentStateType } from "../state";
import { getOpenRouterClient } from "../../core/llm";
import { regenerateWithConstraints } from "./generate";

const FAITHFULNESS_PROMPT = `
You are a faithfulness verifier. Given an answer and source chunks,
check if every factual claim in the answer is supported by the sources.

Answer: {answer}

Source chunks:
{sources}

Respond ONLY with valid JSON:
{"faithfulness_score": 0.0-1.0, "unsupported_claims": ["..."]}
`;

export async function faithfulnessCheck(state: AgentStateType): Promise<Partial<AgentStateType>> {
  const sourcesText = state.parentChunks.map((c: any) => c.text).join("\n---\n");
  const prompt = FAITHFULNESS_PROMPT
    .replace("{answer}", state.finalAnswer)
    .replace("{sources}", sourcesText);

  const client = getOpenRouterClient();
  const response = await client.chat.completions.create({
    model: "google/gemini-2.5-flash",            // or config.LLM_MODEL_ASSESS
    messages: [{ role: "user", content: prompt }],
    temperature: 0,
    response_format: { type: "json_object" }
  });

  const result = JSON.parse(response.choices[0].message.content!);
  let { finalAnswer } = state;

  // Regenerate if score is too low
  if (result.faithfulness_score < 0.85) {
    // Trigger stricter regeneration (add unsupported claims to prompt)
    finalAnswer = await regenerateWithConstraints(state, result.unsupported_claims);
  }

  return { finalAnswer, faithfulnessScore: result.faithfulness_score };
}

import { langfuse } from "../../observability/langfuseClient";
import { AgentStateType } from "../state";
import { hybridSearch } from "../../retrieval/hybridSearch";

export async function retrieve(state: AgentStateType): Promise<Partial<AgentStateType>> {
  const span = langfuse.span({ name: "retrieve_node", input: { query: state.query } });

  const results = await hybridSearch(state.query, state.tenantId, state.collection);

  // Log retrieval metadata to the current Langfuse span
  span.end({
    output: { resultCount: results.length },
    metadata: {
      topScores: results.slice(0, 5).map(r => r.score),
      collection: state.collection
    }
  });

  return { retrievedChunks: results.map(r => r.payload) };
}
